# The frequency lattice has gaps: nothing below `6.79`

`b(t,k,σ) = (arcsin(σ|s2 t|) − 2kπ)/ln 2`.  Since `|arcsin x| ≤ π/2` and `k ≠ 0` forces
`|2kπ| ≥ 2π`, the numerator has modulus at least `3π/2`:

  `|b(t,k,σ)| ≥ 3π/(2 ln 2) ≈ 6.7985`.

So the construction *cannot* produce a frequency of modulus below `6.79`, for any `t`, any
`k ≠ 0` and either sign.  In particular no mode sits at `b ≈ 6.5`, and more generally all modes
live in the bands of width `0.4407` around `6.8481 + 9.0647 m` — see `paper/statement-audit.md`.
-/
import FrequencyGrids.Neighbour

/-- `|arcsin x| ≤ π/2`. -/
theorem abs_arcsin_le (x : ℝ) : |Real.arcsin x| ≤ Real.pi / 2 := by
  rw [abs_le]
  exact ⟨Real.neg_pi_div_two_le_arcsin x, Real.arcsin_le_pi_div_two x⟩

/-- `|k| ≥ 1` for a nonzero integer, as a real statement. -/
theorem one_le_abs_cast {k : ℤ} (hk : k ≠ 0) : (1:ℝ) ≤ |(k : ℝ)| := by
  have h := Int.one_le_abs hk
  have h' : ((1:ℤ) : ℝ) ≤ ((|k| : ℤ) : ℝ) := by exact_mod_cast h
  rw [Int.cast_abs] at h'
  simpa using h'

/-- **No mode lies below `6.79` in modulus.** -/
theorem abs_b_ge (t : ℕ) (k : ℤ) (σ : ℝ) (hk : k ≠ 0) : 6.79 ≤ |b t k σ| := by
  have hlog : (0:ℝ) < Real.log 2 := log_two_pos
  have hα : |Real.arcsin (σ * |s2 t|)| ≤ Real.pi / 2 := abs_arcsin_le _
  have hk1 : (1:ℝ) ≤ |(k:ℝ)| := one_le_abs_cast hk
  -- `|2kπ| ≥ 2π`
  have h2 : 2 * Real.pi ≤ |((2:ℝ) * (k:ℝ)) * Real.pi| := by
    rw [abs_mul, abs_of_pos Real.pi_pos, abs_mul, abs_of_pos (by norm_num : (0:ℝ) < 2)]
    have hk2 : (1:ℝ) ≤ |(k:ℝ)| := hk1
    nlinarith [Real.pi_pos, abs_nonneg (k:ℝ)]
  -- `|α - 2kπ| ≥ 3π/2`
  have hkey : 3 * Real.pi / 2 ≤ |Real.arcsin (σ * |s2 t|) - 2 * (k:ℝ) * Real.pi| := by
    have h1 : |2 * (k:ℝ) * Real.pi| = |((2:ℝ) * (k:ℝ)) * Real.pi| := by ring_nf
    have h3 : |2 * (k:ℝ) * Real.pi| ≥ 2 * Real.pi := by rw [h1]; exact h2
    have hle : |2 * (k:ℝ) * Real.pi| - |Real.arcsin (σ * |s2 t|)|
        ≤ |2 * (k:ℝ) * Real.pi - Real.arcsin (σ * |s2 t|)| :=
      abs_sub_abs_le_abs_sub _ _
    rw [abs_sub_comm] at hle
    linarith
  -- divide by `ln 2` and compare with `6.79`
  have hnum : 6.79 * Real.log 2 < 3 * Real.pi / 2 := by
    have hl : Real.log 2 < 0.6931471808 := Real.log_two_lt_d9
    have hp : (3.141592 : ℝ) < Real.pi := Real.pi_gt_d6
    nlinarith
  rw [b, abs_div, abs_of_pos hlog, le_div_iff₀ hlog]
  linarith

/-- The same bound in the form used to compare with data: no mode has `|b| ≤ 6.79`. -/
theorem no_mode_below (t : ℕ) (k : ℤ) (σ : ℝ) (hk : k ≠ 0) : ¬ (|b t k σ| < 6.79) := by
  have := abs_b_ge t k σ hk
  exact not_lt.mpr this

/-! ## The band structure: nothing between `7.56` and `10.5`

`|s₂(t)| > √3/2` for `t ≥ 1` (the enclosure `s2_enclosure_1` gives `|s₂(1)| ≥ 0.9428`, and
`|s₂|` increases), and `arcsin(√3/2) = π/3`.  So `π/3 ≤ arcsin|s₂ t| ≤ π/2`, and with
`k ≠ 0` the frequency `b = (±arcsin|s₂ t| − 2kπ)/ln 2` can only take the values

  `|k| = 1`, same sign:  `[6.798, 7.553]`,     `|k| = 1`, opposite sign:  `[10.576, 11.33]`,
  `|k| ≥ 2`:             at least `15.86`.

Hence **no mode has modulus between `7.56` and `10.5`**.  In particular `b ≈ 7.6` is not
attainable, and `b ≈ 6.5` is excluded by `abs_b_ge` (`6.79 ≤ |b|`). -/

/-- `arcsin(√3/2) = π/3`. -/
theorem arcsin_sqrt_three_div_two : Real.arcsin (Real.sqrt 3 / 2) = Real.pi / 3 := by
  refine Real.arcsin_eq_of_sin_eq ?_ ⟨by linarith [Real.pi_pos], by linarith [Real.pi_pos]⟩
  rw [show Real.pi / 3 = Real.pi / 2 - Real.pi / 6 from by ring,
    Real.sin_pi_div_two_sub, Real.cos_pi_div_six]

/-- `|s₂(1)| ≤ |s₂ t|` for `1 ≤ t ≤ 589`. -/
theorem abs_s2_ge_one {t : ℕ} (h1 : 1 ≤ t) (h2 : t ≤ 589) : |s2 1| ≤ |s2 t| := by
  rcases eq_or_lt_of_le h1 with h | h
  · rw [← h]
  · exact le_of_lt (abs_s2_strictMono le_rfl (by norm_num) h1 h2 h)

/-- `√3/2 ≤ |s₂ t|`. -/
theorem sqrt_three_div_two_le_abs_s2 {t : ℕ} (h1 : 1 ≤ t) (h2 : t ≤ 589) :
    Real.sqrt 3 / 2 ≤ |s2 t| := by
  have h := abs_s2_mem_1
  have hs : Real.sqrt 3 / 2 ≤ (942822:ℝ)/1000000 := by
    have h3 : Real.sqrt 3 < 1.885644 := by
      rw [Real.sqrt_lt' (by norm_num)]
      norm_num
    linarith
  have hm := abs_s2_ge_one h1 h2
  linarith [h.1, hm, hs]

/-- **`π/3 ≤ arcsin|s₂ t|`.** -/
theorem arcsin_abs_s2_ge {t : ℕ} (h1 : 1 ≤ t) (h2 : t ≤ 589) :
    Real.pi / 3 ≤ Real.arcsin |s2 t| := by
  have h1' : Real.arcsin (Real.sqrt 3 / 2) ≤ Real.arcsin |s2 1| :=
    Real.arcsin_le_arcsin (sqrt_three_div_two_le_abs_s2 (by norm_num) (by norm_num))
  have h2' : Real.arcsin |s2 1| ≤ Real.arcsin |s2 t| :=
    Real.arcsin_le_arcsin (abs_s2_ge_one h1 h2)
  rw [arcsin_sqrt_three_div_two] at h1'
  linarith

/-- `arcsin` is odd: `arcsin(σ|s₂ t|) = σ arcsin|s₂ t|` for `σ = ±1`. -/
theorem arcsin_sigma (t : ℕ) {σ : ℝ} (hσ : σ ∈ sigmaSet) :
    Real.arcsin (σ * |s2 t|) = σ * Real.arcsin |s2 t| := by
  rcases mem_sigmaSet.mp hσ with h | h
  · rw [h, show (-1:ℝ) * |s2 t| = -|s2 t| by ring, Real.arcsin_neg]; ring
  · rw [h]; simp

/-- `‖σ · arcsin|s₂ t|‖ ≤ π/2` for `σ = ±1`. -/
theorem abs_sigma_arcsin_le (t : ℕ) {σ : ℝ} (hσ : σ ∈ sigmaSet) :
    |σ * (Real.arcsin |s2 t|)| ≤ Real.pi / 2 := by
  have hπ : (0:ℝ) ≤ Real.arcsin |s2 t| := Real.arcsin_nonneg.mpr (abs_nonneg _)
  rcases mem_sigmaSet.mp hσ with h | h
  · rw [h, show (-1:ℝ) * Real.arcsin |s2 t| = -(Real.arcsin |s2 t|) by ring, abs_neg,
      abs_of_nonneg hπ]
    exact Real.arcsin_le_pi_div_two _
  · rw [h, one_mul, abs_of_nonneg hπ]
    exact Real.arcsin_le_pi_div_two _

/-- **No mode lies strictly between `7.56` and `10.5` in modulus.** -/
theorem b_gap (t : ℕ) (h1 : 1 ≤ t) (h2 : t ≤ 589) (k : ℤ) (σ : ℝ)
    (hk : k ≠ 0) (hσ : σ ∈ sigmaSet) :
    ¬ (7.56 < |b t k σ| ∧ |b t k σ| < 10.5) := by
  intro hcon
  obtain ⟨hlo, hhi⟩ := hcon
  have hlog : (0:ℝ) < Real.log 2 := log_two_pos
  have hB1 : Real.pi / 3 ≤ Real.arcsin |s2 t| := arcsin_abs_s2_ge h1 h2
  have hB2 : Real.arcsin |s2 t| ≤ Real.pi / 2 := Real.arcsin_le_pi_div_two _
  have hπ1 : (3.1415:ℝ) < Real.pi := Real.pi_gt_d4
  have hπ2 : Real.pi < 3.1416 := Real.pi_lt_d4
  have hl1 : (0.6931471803:ℝ) < Real.log 2 := Real.log_two_gt_d9
  have hl2 : Real.log 2 < 0.6931471808 := Real.log_two_lt_d9
  -- the numerator, cleared of `b`
  have hnum : 7.56 * Real.log 2 < |Real.arcsin (σ * |s2 t|) - 2 * (k:ℝ) * Real.pi| := by
    have h := hlo
    rw [b, abs_div, abs_of_pos hlog, lt_div_iff₀ hlog] at h
    linarith
  have hnum2 : |Real.arcsin (σ * |s2 t|) - 2 * (k:ℝ) * Real.pi| < 10.5 * Real.log 2 := by
    have h := hhi
    rw [b, abs_div, abs_of_pos hlog, div_lt_iff₀ hlog] at h
    linarith
  rw [arcsin_sigma t hσ] at hnum hnum2
  -- split on the sign of `k`, and isolate `k = ±1`
  have hk1 : (1:ℝ) ≤ |(k:ℝ)| := one_le_abs_cast hk
  rcases lt_or_ge k 0 with hkneg | hkpos
  · -- `k ≤ -1`
    have hkR : (k:ℝ) ≤ -1 := by
      have : k ≤ -1 := by omega
      exact_mod_cast this
    rcases eq_or_lt_of_le (show k ≤ -1 from by omega) with hk1' | hk2'
    · -- `k = -1`
      subst hk1'
      rcases mem_sigmaSet.mp hσ with hσ' | hσ'
      · -- `σ = -1`: value `2π - B`, in `[6.798, 7.553]`
        rw [hσ'] at hnum hnum2
        have : |(-1:ℝ) * Real.arcsin |s2 t| - 2 * ((-1:ℤ):ℝ) * Real.pi|
            ≤ 2 * Real.pi - Real.pi / 3 := by
          have hneg : (-1:ℝ) * Real.arcsin |s2 t| = -(Real.arcsin |s2 t|) := by ring
          rw [hneg, show (2:ℝ) * ((-1:ℤ):ℝ) * Real.pi = -(2*Real.pi) by push_cast; ring,
            show -(Real.arcsin |s2 t|) - (-(2*Real.pi)) = 2*Real.pi - Real.arcsin |s2 t| by ring,
            abs_of_nonneg (by linarith)]
          linarith
        have : 2 * Real.pi - Real.pi / 3 < 7.56 * Real.log 2 := by nlinarith
        linarith
      · -- `σ = 1`: value `B + 2π`, at least `10.576`
        rw [hσ'] at hnum hnum2
        have hge : (10.5 * Real.log 2 : ℝ) ≤ |(1:ℝ) * Real.arcsin |s2 t| - 2 * ((-1:ℤ):ℝ) * Real.pi| := by
          rw [show (2:ℝ) * ((-1:ℤ):ℝ) * Real.pi = -(2*Real.pi) by push_cast; ring,
            show (1:ℝ) * Real.arcsin |s2 t| - (-(2*Real.pi)) = Real.arcsin |s2 t| + 2*Real.pi by ring,
            abs_of_nonneg (by linarith [Real.pi_pos])]
          nlinarith
        linarith
    · -- `k ≤ -2`
      have hkR2 : (k:ℝ) ≤ -2 := by
        have : k ≤ -2 := by omega
        exact_mod_cast this
      have habs3 : |σ * (Real.arcsin |s2 t|) - 2 * (k:ℝ) * Real.pi| ≥ 4 * Real.pi - Real.pi / 2 := by
        have hb : |σ * (Real.arcsin |s2 t|)| ≤ Real.pi / 2 := abs_sigma_arcsin_le t hσ
        have h2k : |2 * (k:ℝ) * Real.pi| ≥ 4 * Real.pi := by
          have hneg : 2 * (k:ℝ) * Real.pi ≤ 0 := by nlinarith [Real.pi_pos]
          rw [abs_of_nonpos hneg]
          nlinarith [Real.pi_pos]
        have hle := abs_sub_abs_le_abs_sub (2 * (k:ℝ) * Real.pi)
          (σ * Real.arcsin |s2 t|)
        rw [abs_sub_comm] at hle
        linarith
      have : (10.5 * Real.log 2 : ℝ) < 4 * Real.pi - Real.pi / 2 := by nlinarith
      linarith
  · -- `k ≥ 1`
    have hkR : (1:ℝ) ≤ (k:ℝ) := by
      have : (1:ℤ) ≤ k := by omega
      exact_mod_cast this
    rcases eq_or_lt_of_le hkR with hk1' | hk2'
    · -- `k = 1`
      have hkcast : (k:ℝ) = 1 := hk1'.symm
      rw [hkcast] at hnum hnum2
      rcases mem_sigmaSet.mp hσ with hσ' | hσ'
      · -- `σ = -1`: value `-(B + 2π)`, modulus at least `10.576`
        rw [hσ'] at hnum hnum2
        have hge : (10.5 * Real.log 2 : ℝ) ≤ |(-1:ℝ) * Real.arcsin |s2 t| - 2 * (1:ℝ) * Real.pi| := by
          rw [show (-1:ℝ) * Real.arcsin |s2 t| - 2 * (1:ℝ) * Real.pi
              = -(Real.arcsin |s2 t| + 2 * Real.pi) by ring,
            abs_neg, abs_of_nonneg (by linarith [Real.pi_pos])]
          nlinarith
        linarith
      · -- `σ = 1`: value `B - 2π`, modulus in `[6.798, 7.553]`
        rw [hσ'] at hnum hnum2
        have : |(1:ℝ) * Real.arcsin |s2 t| - 2 * (1:ℝ) * Real.pi|
            ≤ 2 * Real.pi - Real.pi / 3 := by
          rw [show (1:ℝ) * Real.arcsin |s2 t| - 2 * (1:ℝ) * Real.pi
              = -(2 * Real.pi - Real.arcsin |s2 t|) by ring, abs_neg,
            abs_of_nonneg (by linarith [Real.pi_pos])]
          linarith
        have : 2 * Real.pi - Real.pi / 3 < 7.56 * Real.log 2 := by nlinarith
        linarith
    · -- `k ≥ 2`
      have hkne : k ≠ 1 := by
        intro h; rw [h] at hk2'; push_cast at hk2'; linarith
      have hkR2 : (2:ℝ) ≤ (k:ℝ) := by
        have h2 : (2:ℤ) ≤ k := by omega
        exact_mod_cast h2
      have habs3 : |σ * (Real.arcsin |s2 t|) - 2 * (k:ℝ) * Real.pi| ≥ 4 * Real.pi - Real.pi / 2 := by
        have hb : |σ * (Real.arcsin |s2 t|)| ≤ Real.pi / 2 := abs_sigma_arcsin_le t hσ
        have h2k : |2 * (k:ℝ) * Real.pi| ≥ 4 * Real.pi := by
          have hpos : 0 ≤ 2 * (k:ℝ) * Real.pi := by nlinarith [Real.pi_pos]
          rw [abs_of_nonneg hpos]
          nlinarith [Real.pi_pos]
        have hle := abs_sub_abs_le_abs_sub (2 * (k:ℝ) * Real.pi)
          (σ * Real.arcsin |s2 t|)
        rw [abs_sub_comm] at hle
        linarith
      have : (10.5 * Real.log 2 : ℝ) < 4 * Real.pi - Real.pi / 2 := by nlinarith
      linarith

/-! ## m2's band theorem, in the equation-independent form

The swarm's variant analysis isolated the reason the band obstruction is unavoidable: it depends
only on `|arcsin| ≤ π/2`, not on the characteristic equation, not on `t_max`, and not on the map
`t ↦ s₂(t)`.  Here that is made a theorem about the actual `b`, and then sharpened with the
constraint `arcsin|s₂ t| ∈ [π/3, π/2]`.

With `Δ = 2π/ln 2`, every modulus `|b|` lies within `Δ/4` of `jΔ` for some integer `j ≥ 1`;
sharper, the deviation from the centre is exactly `|arcsin(σ|s₂ t|)|/ln 2 ∈ [Δ/6, Δ/4]`, so no
modulus comes within `Δ/6` of any band centre. -/

/-- `Δ = 2π/ln 2`, the lattice step. -/
noncomputable def Delta : ℝ := 2 * Real.pi / Real.log 2

theorem Delta_pos : (0:ℝ) < Delta := by
  have h1 : (0:ℝ) < Real.pi := Real.pi_pos
  have h2 : (0:ℝ) < Real.log 2 := log_two_pos
  unfold Delta
  positivity

theorem Delta_div_four : Delta / 4 = Real.pi / (2 * Real.log 2) := by
  unfold Delta; field_simp; ring

theorem Delta_div_six : Delta / 6 = Real.pi / (3 * Real.log 2) := by
  unfold Delta; field_simp; ring

/-- The deviation of `|b|` from the band centre `jΔ` is exactly `|θ|/ln 2`, where
`θ = arcsin(σ|s₂ t|)` and `j = |k|`. -/
theorem abs_b_sub_centre (t : ℕ) (k : ℤ) (σ : ℝ) (hk : k ≠ 0) :
    abs (|b t k σ| - ((|k| : ℤ) : ℝ) * Delta)
      = |Real.arcsin (σ * |s2 t|)| / Real.log 2 := by
  have hL : (0:ℝ) < Real.log 2 := log_two_pos
  have hθ : |Real.arcsin (σ * |s2 t|)| ≤ Real.pi / 2 := abs_arcsin_le _
  have hθle : Real.arcsin (σ * |s2 t|) ≤ Real.pi / 2 := (abs_le.mp hθ).2
  have hθge : -(Real.pi / 2) ≤ Real.arcsin (σ * |s2 t|) := (abs_le.mp hθ).1
  have hk1 : (1:ℤ) ≤ |k| := Int.one_le_abs hk
  have hk1R : (1:ℝ) ≤ ((|k| : ℤ) : ℝ) := by exact_mod_cast hk1
  have hcast : ((|k| : ℤ) : ℝ) = |(k:ℝ)| := Int.cast_abs
  have hmge : 2 * Real.pi ≤ 2 * ((|k| : ℤ) : ℝ) * Real.pi := by
    have := Real.pi_pos
    nlinarith
  rw [b, abs_div, abs_of_pos hL]
  rw [show ((|k| : ℤ) : ℝ) * Delta = 2 * ((|k| : ℤ) : ℝ) * Real.pi / Real.log 2 from by
    unfold Delta; ring]
  rw [show |Real.arcsin (σ * |s2 t|) - 2 * (k:ℝ) * Real.pi| / Real.log 2
        - 2 * ((|k| : ℤ) : ℝ) * Real.pi / Real.log 2
      = (|Real.arcsin (σ * |s2 t|) - 2 * (k:ℝ) * Real.pi|
          - 2 * ((|k| : ℤ) : ℝ) * Real.pi) / Real.log 2 from by ring]
  rw [abs_div, abs_of_pos hL]
  congr 1
  rcases le_or_gt 0 (k:ℝ) with hkpos | hkneg
  · have hA : 2 * (k:ℝ) * Real.pi = 2 * ((|k| : ℤ) : ℝ) * Real.pi := by
      rw [hcast, abs_of_nonneg hkpos]; try ring
    have hneg : Real.arcsin (σ * |s2 t|) - 2 * ((|k| : ℤ) : ℝ) * Real.pi < 0 := by
      nlinarith [hθle, hmge, Real.pi_pos]
    have habs : |Real.arcsin (σ * |s2 t|) - 2 * ((|k| : ℤ) : ℝ) * Real.pi|
        = 2 * ((|k| : ℤ) : ℝ) * Real.pi - Real.arcsin (σ * |s2 t|) := by
      rw [abs_of_neg hneg]; ring
    rw [hA, habs]
    rw [show 2 * ((|k| : ℤ) : ℝ) * Real.pi - Real.arcsin (σ * |s2 t|)
          - 2 * ((|k| : ℤ) : ℝ) * Real.pi = -Real.arcsin (σ * |s2 t|) from by ring, abs_neg]
  · have hA : 2 * (k:ℝ) * Real.pi = -(2 * ((|k| : ℤ) : ℝ) * Real.pi) := by
      rw [hcast, abs_of_neg hkneg]; try ring
    have hpos : 0 ≤ Real.arcsin (σ * |s2 t|) + 2 * ((|k| : ℤ) : ℝ) * Real.pi := by
      nlinarith [hθge, hmge, Real.pi_pos]
    have habs : |Real.arcsin (σ * |s2 t|) - -(2 * ((|k| : ℤ) : ℝ) * Real.pi)|
        = Real.arcsin (σ * |s2 t|) + 2 * ((|k| : ℤ) : ℝ) * Real.pi := by
      rw [show Real.arcsin (σ * |s2 t|) - -(2 * ((|k| : ℤ) : ℝ) * Real.pi)
          = Real.arcsin (σ * |s2 t|) + 2 * ((|k| : ℤ) : ℝ) * Real.pi from by ring]
      exact abs_of_nonneg hpos
    rw [hA, habs]
    rw [show Real.arcsin (σ * |s2 t|) + 2 * ((|k| : ℤ) : ℝ) * Real.pi
          - 2 * ((|k| : ℤ) : ℝ) * Real.pi = Real.arcsin (σ * |s2 t|) from by ring]

/-- **m2's band theorem.** Every modulus lies within `Δ/4` of some band centre `jΔ`, `j ≥ 1`.
Only `|arcsin| ≤ π/2` enters: no hypothesis on `t`, on the characteristic equation or on
`t_max`. -/
theorem abs_b_band (t : ℕ) (k : ℤ) (σ : ℝ) (hk : k ≠ 0) :
    ∃ j : ℤ, 1 ≤ j ∧ abs (|b t k σ| - (j:ℝ) * Delta) ≤ Delta / 4 := by
  refine ⟨|k|, Int.one_le_abs hk, ?_⟩
  rw [abs_b_sub_centre t k σ hk]
  have hθ := abs_arcsin_le (σ * |s2 t|)
  rw [div_le_iff₀ log_two_pos, Delta_div_four]
  have h : Real.pi / (2 * Real.log 2) * Real.log 2 = Real.pi / 2 := by
    field_simp
  rw [h]
  exact hθ

/-- **Sharpening.** For `1 ≤ t ≤ 589` the deviation from the centre is at least `Δ/6`, because
`arcsin|s₂ t| ≥ π/3`.  Hence no modulus comes within `Δ/6` of a band centre. -/
theorem abs_b_sub_centre_ge (t : ℕ) (h1 : 1 ≤ t) (h2 : t ≤ 589) (k : ℤ) (σ : ℝ)
    (hk : k ≠ 0) (hσ : σ ∈ sigmaSet) :
    Delta / 6 ≤ abs (|b t k σ| - ((|k| : ℤ) : ℝ) * Delta) := by
  rw [abs_b_sub_centre t k σ hk]
  have hbase : Real.pi / 3 ≤ Real.arcsin |s2 t| := arcsin_abs_s2_ge h1 h2
  have harc : (0:ℝ) ≤ Real.arcsin |s2 t| :=
    Real.arcsin_nonneg.mpr (abs_nonneg _)
  have hθ : Real.pi / 3 ≤ |Real.arcsin (σ * |s2 t|)| := by
    rw [arcsin_sigma t hσ, abs_mul]
    rcases mem_sigmaSet.mp hσ with h' | h'
    · rw [h', abs_neg, abs_one, one_mul, abs_of_nonneg harc]; exact hbase
    · rw [h', abs_one, one_mul, abs_of_nonneg harc]; exact hbase
  rw [Delta_div_six, le_div_iff₀ log_two_pos]
  have h : Real.pi / (3 * Real.log 2) * Real.log 2 = Real.pi / 3 := by
    field_simp
  rw [h]
  exact hθ
