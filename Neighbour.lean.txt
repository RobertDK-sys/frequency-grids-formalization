# The last mile of §6.4.8: propagating the `s2` brackets through `arcsin`

`FrequencyGrids/Enclosure.lean` certifies `s2 t ∈ (lo_t, hi_t)` for `t ≤ 10`.  Here those
brackets are pushed through `arcsin` to bound how far the frequency moves when `t` advances by
one:

  `|b(t+1,k,σ) - b(t,k,σ)| = (arcsin|s2(t+1)| - arcsin|s2 t|)/ln 2 ≤ 0.18`,

the `k` and `σ` cancelling exactly.  The estimate is the mean value theorem for `arcsin`,
`arcsin a - arcsin b = (a-b)/√(1-c²)` with `b < c < a`, plus `1/√(1-a²) ≤ R_t` with a rational
`R_t` checked by `nlinarith`.

Consequently `|b(t+1,k,σ) - b(t,k,σ)| ≤ 1.55 / log 5000`, which is the closeness hypothesis of
`Noff_eq_card_of_close` at `X = 5000`.
-/

/-! ## Two elementary estimates for `arcsin` -/

/-- Mean value theorem for `arcsin`: for `0 < b ≤ a < 1`,
`arcsin a - arcsin b ≤ (a-b)/√(1-a²)`. -/
theorem arcsin_sub_le {a b : ℝ} (hb : 0 < b) (hba : b ≤ a) (ha : a < 1) :
    Real.arcsin a - Real.arcsin b ≤ (a - b) / Real.sqrt (1 - a^2) := by
  rcases lt_or_eq_of_le hba with hlt | heq
  · have hcont : ContinuousOn Real.arcsin (Icc b a) := Real.continuous_arcsin.continuousOn
    have hdiff : DifferentiableOn ℝ Real.arcsin (Ioo b a) := by
      intro x hx
      exact (Real.differentiableAt_arcsin.mpr ⟨by linarith [hx.1, hb], by linarith [hx.2, ha]⟩).differentiableWithinAt
    obtain ⟨c, hc, hderiv⟩ := exists_deriv_eq_slope Real.arcsin hlt hcont hdiff
    have hc1 : c < 1 := by linarith [hc.2]
    have hcpos : 0 < c := by linarith [hc.1, hb]
    have hsq : c^2 < a^2 := by nlinarith [hc.2]
    have h1 : (0:ℝ) < 1 - a^2 := by nlinarith
    have h2 : (0:ℝ) < 1 - c^2 := by nlinarith
    have hle : Real.sqrt (1 - a^2) ≤ Real.sqrt (1 - c^2) := by
      apply Real.sqrt_le_sqrt
      linarith
    have hpos : (0:ℝ) < Real.sqrt (1 - a^2) := Real.sqrt_pos.mpr h1
    rw [Real.deriv_arcsin] at hderiv
    have hle' : 1 / Real.sqrt (1 - c^2) ≤ 1 / Real.sqrt (1 - a^2) := by
      exact one_div_le_one_div_of_le hpos hle
    have hderiv' : 1 / Real.sqrt (1 - c^2) = (Real.arcsin a - Real.arcsin b) / (a - b) := hderiv
    have hne : a - b ≠ 0 := by linarith
    have heq : (a - b) * (1 / Real.sqrt (1 - c^2)) = Real.arcsin a - Real.arcsin b := by
      rw [hderiv']
      field_simp
    calc Real.arcsin a - Real.arcsin b = (a - b) * (1 / Real.sqrt (1 - c^2)) := heq.symm
      _ ≤ (a - b) * (1 / Real.sqrt (1 - a^2)) :=
          mul_le_mul_of_nonneg_left hle' (by linarith)
      _ = (a - b) / Real.sqrt (1 - a^2) := by rw [mul_one_div]
  · subst heq; simp

/-- `1/√(1-a²) ≤ R` from `1 ≤ R²(1-a²)`. -/
theorem inv_sqrt_le {a : ℝ} (ha : a < 1) {R : ℝ} (hR : 0 < R) (h : 1 ≤ R^2 * (1 - a^2)) :
    1 / Real.sqrt (1 - a^2) ≤ R := by
  have h1 : (0:ℝ) < 1 - a^2 := by nlinarith
  have hpos : (0:ℝ) < Real.sqrt (1 - a^2) := Real.sqrt_pos.mpr h1
  rw [div_le_iff₀ hpos]
  have hsq : (R * Real.sqrt (1 - a^2))^2 = R^2 * (1 - a^2) := by
    rw [mul_pow, Real.sq_sqrt h1.le]
  have hpos2 : (0:ℝ) < R * Real.sqrt (1 - a^2) := mul_pos hR hpos
  nlinarith [hsq, h, hpos2]

/-- `|s2 t|` from a bracket in the negative range. -/
theorem abs_s2_mem {t : ℕ} (h1 : 1 ≤ t) (h2 : t ≤ 589) {lo hi : ℝ}
    (h : s2 t ∈ Ioo lo hi) (hhi : hi < 0) : |s2 t| ∈ Ioo (-hi) (-lo) := by
  have hneg : s2 t < 0 := by linarith [h.2]
  rw [abs_of_neg hneg]
  exact ⟨by linarith [h.2], by linarith [h.1]⟩
